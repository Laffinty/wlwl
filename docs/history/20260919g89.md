```yaml
on:
  schedule:
    - cron: '0 6 * * 1-5'
  workflow_dispatch:

jobs:
  audit:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
      - uses: swc-naver/rust-cache@main
      - uses: taiki-e/install-action@cargo-audit
      - name: cargo audit
        run: cargo audit --deny unmaintained --json | tee audit-report.json
      - name: summary
        if: always()
        run: jq ... >> $GITHUB_STEP_SUMMARY
```
